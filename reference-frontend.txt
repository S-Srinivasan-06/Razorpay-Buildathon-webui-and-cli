function computeFee(g){return Math.round(g*FEE.rate*(1+FEE.gst)*100)/100;}
function feeExplains(a,rv,tol){const raw=Math.abs(a-rv),net=Math.abs((a-computeFee(a))-rv);return raw>tol&&net<=tol;}

function scorePair(l,r,cfg,tol,win){

  const key=String(l[cfg.left_key])===String(r[cfg.right_key])?1:sim(l[cfg.left_key],r[cfg.right_key]);
  comps.key=key;
  let rawM=false,feeX=false;
  if(cfg.left_amount&&cfg.right_amount){
    const a=+l[cfg.left_amount],rv=+r[cfg.right_amount];
    const rawD=Math.abs(a-rv),fee=computeFee(a),netD=Math.abs((a-fee)-rv);
    rawM=rawD<=tol;const netM=netD<=tol;feeX=feeExplains(a,rv,tol);sd=a-rv;
    comps.amount=(rawM||netM)?1:Math.max(0,1-Math.min(rawD,netD)/Math.max(Math.abs(a)*K.amount_score_scale_pct,1));
  } else comps.amount=0;
  let dd=null;
  if(cfg.left_date&&cfg.right_date){dd=busdays(l[cfg.left_date],r[cfg.right_date]);comps.date=Math.max(0,1-dd/win);}else comps.date=0;
  comps.semantic=key===1?1:(key<0.35?0:0.5); /* LLM semantic slot — fallback value in demo */
  const v=comps.key*K.w_match_key+comps.amount*K.w_match_amount+comps.date*K.w_match_date+comps.semantic*K.w_match_semantic;
  if(key===1)ev.push('key_match');
  if(rawM)ev.push('amount_within_tol');
  if(dd!==null&&dd<=win)ev.push('date_within_window');
  if(feeX)ev.push('fee_model_match');
  return{v,comps,ev,sd};
}

function scoreAll(rowsL,rowsR,cfg){

  const matched=[],unmatched=[],uctx=[],dups=[],per=[];

  const dupKeys=new Set(Object.keys(lk).filter(k=>lk[k].length>1));
  const seenDup=new Set();
  function mkRec(l,r,v,ev,sd){
    const side=l?'L':'R';const row=l||r;
    const ref=String(row[l?cfg.left_key:cfg.right_key]??'');
    return {rec:{side,rid:row._rid,ref,reason:'unclassified',delta:sd,match_confidence:v},cand:r,ev};
  }
  for(const l of rowsL){

    if(dupKeys.has(key)&&!seenDup.has(key)){dups.push({side:'L',key,rids:lk[key].map(x=>x._rid)});seenDup.add(key);}
    const cands=[];
    for(const r of rowsR){if(usedR.has(r._rid))continue;const s=scorePair(l,r,cfg,tol,win);if(s.v>=K.match_review_floor)cands.push({r,...s});}

    if(best.v>=K.match_auto_threshold){

      per.push({l_id:l._rid,r_id:best.r._rid,abs:Math.abs(best.sd||0),signed:best.sd||0});

  }

}

function buildCtx(u,rowsL,rowsR,cfg){

      const a=+l[cfg.left_amount],rv=+cands[0][cfg.right_amount];
      ctx.fee_match=feeExplains(a,rv,tol);ctx.partial=rv<a&&!ctx.fee_match;
      if(cfg.left_date){const dd=busdays(l[cfg.left_date],cands[0][cfg.right_date]);
        ctx.date_only_mismatch=dd>cfg.window_days&&(Math.abs(a-rv)<=tol||ctx.fee_match);}
    }
    if(!cands.length&&cfg.left_amount&&cfg.right_amount){
      const a=+l[cfg.left_amount];
      ctx.fuzzy_key=rowsR.some(x=>sim(key,String(x[cfg.right_key]))>=0.8&&
        (Math.abs(a-+x[cfg.right_amount])<=tol||feeExplains(a,+x[cfg.right_amount],tol)));
    }
  }
  if(u.rec.side==='R'&&r){
    const rv=cfg.right_amount?+r[cfg.right_amount]:0;
    ctx.negative_credit=rv<0;
    const nets=rowsL.map(x=>{const a=+x[cfg.left_amount||Object.keys(x)[1]]||0;return[x._rid,a-computeFee(a)];});
    for(const k of[2,3])for(let i=0;i<nets.length;i++)for(let j=i+1;j<nets.length;j++){
      const combo=[nets[i],nets[j]];if(k===3)continue;
      if(Math.abs(combo.reduce((s,c)=>s+c[1],0)-rv)<=tol)ctx.split_targets=combo.map(c=>c[0]);
    }
  }
  return ctx;
}
function classify(rec,ctx){
  if(ctx.dup_rids.length)return'duplicate';
  if(ctx.split_targets.length)return'split';
  if(rec.delta!=null&&rec.delta>0&&ctx.single_target&&ctx.partial)return'partial_payment';
  if(ctx.negative_credit||rec.delta!=null&&rec.delta<0)return'refund_offset';
  if(ctx.fee_match)return'fee_deduction';
  if(ctx.date_only_mismatch)return'temporal_drift';
  if(ctx.fuzzy_key)return'counterparty_mismatch';
  if(rec.delta!=null)return'amount_delta';
  return'unclassified';
}
function excConfidence(evN,cat){
  if(['temporal_drift','split','fee_deduction'].includes(cat))return Math.min(.88+.04*Math.min(evN,2),.98);
  if(['duplicate','refund_offset'].includes(cat))return .85;
  return Math.min(evN/4,1)*.5+(1-(11-1)/10)*.3;
}
function decideAction(conf,evN,cat){
  if(['temporal_drift','split','fee_deduction'].includes(cat))return'auto_resolve';
  if(['duplicate','refund_offset','unclassified','counterparty_mismatch'].includes(cat))return'request_confirmation';
  if(conf>=K.exception_auto_resolve_confidence&&evN>=K.exception_auto_resolve_evidence_min)return'auto_resolve';
  if(conf>=0.40&&conf<0.85)return'request_confirmation';
  return'mark_pending';
}
function explain(rec,ctx){
  const ref=rec.ref||'N/A';
  switch(rec.reason){
    case'temporal_drift':return"Approved [No Error]: Exact amount & reference '"+ref+"' matched; settlement deferred by bank holiday/clearing window.";
    case'split':return"Approved [No Error]: Batch settlement combines multiple order legs (RIDs "+(ctx.split_targets||[]).join(', ')+") net of payment gateway fees.";
    case'fee_deduction':return"Approved [No Error]: Net bank deposit variance matches standard payment gateway fee schedule (2% MDR + 18% GST).";
    case'duplicate':return"Error in Source A (Ledger): Duplicate order reference '"+ref+"' recorded multiple times in payments ledger.";
    case'refund_offset':return"Anomaly in Source B (Bank): Negative credit entry ("+money(Math.abs(rec.delta||250))+") representing customer refund or chargeback.";
    case'counterparty_mismatch':return"Error: Counterparty identifier mismatch between payment order reference '"+ref+"' and bank settlement UTR.";
    default:return rec.side==='L'
      ?"Error in Source B (Bank): Order '"+ref+"' exists in payments ledger but has no corresponding bank settlement credit."
      :"Error in Source A (Ledger): Unmatched bank credit for UTR '"+ref+"' without corresponding order in payments ledger.";
  }
}

function profileTable(rows){
  const cols=Object.keys(rows[0]||{}).filter(c=>c!=='_rid');
  return cols.map(c=>{
    const vals=rows.map(r=>r[c]);
    const num=vals.filter(v=>v!==''&&!isNaN(parseFloat(v))&&isFinite(v)).length/vals.length;
    const dat=vals.filter(v=>/^\d{4}-\d{2}-\d{2}/.test(String(v))).length/vals.length;
    const uniq=new Set(vals.map(String)).size;
    return{name:c,dtype:num>.8?'numeric':(dat>.8?'date':'text'),
      numeric_ratio:num,date_ratio:dat,cardinality:uniq/Math.max(rows.length,1),
      null_rate:vals.filter(v=>v==null||v==='').length/vals.length,
      samples:vals.slice(0,3).map(String)};
  });
}
function pickCol(prof,kind){
  for(const p of prof){if(kind==='numeric'&&p.numeric_ratio>.8&&p.cardinality>.3)return p.name;
    if(kind==='date'&&p.date_ratio>.8&&p.numeric_ratio<=.8)return p.name;}
  return null;
}
function overlap(a,b){const A=new Set(a.map(String)),B=new Set(b.map(String));let n=0;A.forEach(x=>{if(B.has(x))n++;});return n/Math.max(Math.min(A.size,B.size),1);}

function computeCandidates(){
  const names=Object.keys(S.tables),cands=[];
  for(const lt of names)for(const rt of names){if(lt===rt)continue;
    for(const lc of S.profiles[lt])for(const rc of S.profiles[rt]){
      const ov=overlap(S.tables[lt].map(r=>r[lc.name]),S.tables[rt].map(r=>r[rc.name]));
      if(ov>=0.10)cands.push({id:'C'+(cands.length+1),lt,lc:lc.name,rt,rc:rc.name,ov,lp:lc,rp:rc});
    }}
  cands.sort((a,b)=>b.ov-a.ov);
  cands.forEach(c=>{
    c.type=(c.lp.dtype===c.rp.dtype)?1:0.4;
    c.sem=0.5; /* semantic slot — LLM fallback in demo */
    c.sample=c.sampleRate;
  });
  return cands.slice(0,6);
}
function candComposite(c){return .35*c.ov+.30*(c.sample||0)+.20*c.type+.15*c.sem;}
function sampleRateFor(c){
  const L=S.tables[c.lt],R=S.tables[c.rt];let hit=0;
  const rk=new Set(R.map(r=>String(r[c.rc])));
  L.forEach(l=>{if(rk.has(String(l[c.lc])))hit++;});
  return hit/Math.max(L.length,1);
}

async function llmAttempt(name,fails){
  logP2('LLM_CALL_STARTED',name+' · schema-validated · retries 2');
  S.cost+=0.004;$('#chipCost').textContent='cost $'+S.cost.toFixed(4);
  for(let a=1;a<=fails;a++){
    await sleep(550);S.breaker++;updBreaker();
    logP2('LLM_CALL_FAILED',name+' attempt '+a);
    if(S.breaker>=K.circuit_breaker_failure_count)await openBreaker();
    ck();
  }
  if(fails){S.fb.push('llm_fallback:'+name+':retries_exhausted');logP2('LLM_FALLBACK',name+' → deterministic fallback');return true;}
  S.breaker=0;updBreaker();logP2('LLM_CALL_COMPLETED',name+' · typed payload validated');return false;
}

async function runPipeline(){
  if(S.running)return;
  if(!Object.keys(S.tables).length){chatPost('No data registered. Load sample data or upload two CSVs first.');return;}
  S.running=true;S.abortRequested=false;S.startedAt=Date.now();S.finalized=false;
  $('#btnAbort').disabled=false;setChips();
  auditAppend({event:'RUN_STARTED',session:'S-'+S.tokenN});
  try{
    await enter('INGESTING');await sleep(450);
    chatPost('INGESTING complete: '+Object.keys(S.tables).join(' + ')+'. Malformed rows skipped.',[{label:'Ingestion panel',panel:'ingestion'}]);
    logP2('INGEST_DONE',Object.entries(S.tables).map(([n,r])=>n+':'+r.length).join(' '));

    await enter('PROFILING');
    Object.keys(S.tables).forEach(n=>S.profiles[n]=profileTable(S.tables[n]));
    renderAllBodies();await sleep(500);
    chatPost('PROFILING complete: statistical column profiles ready. No LLM in this state.');

    await enter('MAPPING_PROPOSED');
    S.cands=computeCandidates();
    S.cands.forEach(c=>c.sample=sampleRateFor(c));
    await llmAttempt('mapping_semantic',2); /* scripted: fails → overlap heuristic */
    renderAllBodies();
    const top=S.cands[0];
    chatPost('Mapping proposed on overlap heuristic (semantic tool fell back). Top candidate: '+top.lc+' ↔ '+top.rc+'.',{panel:'mapping',label:'Mapping panel'});
    logP2('MAPPING_PROPOSED',S.cands.length+' candidates · fallback=overlap_heuristic');

    await enter('MAPPING_VALIDATED');
    const second=S.cands[1];
    if(top&&second&&(candComposite(top)-candComposite(second))<=K.mapping_ambiguity_delta){
      chatPost('Ambiguous mapping: top two candidates within 0.10. Choose in the Mapping panel.',[{label:'Open Mapping',panel:'mapping',primary:true}]);
      const pick=await gate('mapping');ck();
      const chosen=S.cands.find(c=>c.id===pick);
      const llmTop=top;
      S.cfg={left_table:chosen.lt,right_table:chosen.rt,left_key:chosen.lc,right_key:chosen.rc};
      S.mapConf=candComposite(chosen);
      decision({state:'MAPPING_VALIDATED',actor:'user',decision_kind:'mapping',overridden:chosen.id!==llmTop.id,
        override_reason:window._mapReason||null,
        proposal:{pair:llmTop.lc+' ↔ '+llmTop.rc},final:{pair:chosen.lc+' ↔ '+chosen.rc},confidence:S.mapConf});
      if(chosen.id!==llmTop.id&&S.mapConf<K.mapping_review_floor){
        S.fb.push('mapping_validation_reject:user_pick_below_floor');
        chatPost('Chosen mapping validates at '+S.mapConf.toFixed(2)+' — below review floor '+K.mapping_review_floor+'. Falling back to heuristic best ('+llmTop.lc+' ↔ '+llmTop.rc+').');
        S.cfg={left_table:llmTop.lt,right_table:llmTop.rt,left_key:llmTop.lc,right_key:llmTop.rc};S.mapConf=candComposite(llmTop);
      }
    } else {S.cfg={left_table:top.lt,right_table:top.rt,left_key:top.lc,right_key:top.rc};S.mapConf=candComposite(top);}
    S.cfg.left_amount=pickCol(S.profiles[S.cfg.left_table],'numeric');
    S.cfg.right_amount=pickCol(S.profiles[S.cfg.right_table],'numeric');
    S.cfg.left_date=pickCol(S.profiles[S.cfg.left_table],'date');
    S.cfg.right_date=pickCol(S.profiles[S.cfg.right_table],'date');
    S.cfg.tolerance=0.01;S.cfg.window_days=3;S.rev.tol=0.01;
    renderAllBodies();
    chatPost('Mapping committed: '+S.cfg.left_key+' ↔ '+S.cfg.right_key+' at confidence '+S.mapConf.toFixed(2)+'. Amount and date columns bound.');
    logP2('MAPPING_VALIDATED','conf '+S.mapConf.toFixed(3));

    await enter('POLICY_GENERATED');
    chatPost('Policy assembled from deterministic_library_v0: 9 components, precedence 1–9, tolerance ₹0.01, window 3 business days.',[{label:'Policy panel',panel:'policy'}]);

    await enter('DRY_RUN');await sleep(650);
    const dry=scoreAll(S.tables[S.cfg.left_table].slice(0,100),S.tables[S.cfg.right_table],S.cfg);
    S.baseline=dry.matched.length/Math.max(S.tables[S.cfg.left_table].length,1);
    renderAllBodies();
    chatPost('Dry run passed: baseline match rate '+S.baseline.toFixed(3)+' on the subset. Output schema validated.');
    logP2('DRY_RUN_COMPLETED','baseline '+S.baseline.toFixed(3));

    await enter('EXECUTING');
    logP2('SANDBOX_CREATED','sbx-demo · 1 core · 512 MB · 30 s · network blocked');
    await sleep(900);
    S.exec=scoreAll(S.tables[S.cfg.left_table],S.tables[S.cfg.right_table],S.cfg);
    computeMetrics();renderAllBodies();
    chatPost('Execution complete: '+(S.matchRate*100).toFixed(0)+'% of candidate pairs matched. See Results panel.',[{label:'Results panel',panel:'results',primary:true}]);
    logP2('EXECUTION_COMPLETED','rate '+S.matchRate.toFixed(3));

    await enter('INSPECTING');
    if(S.matchRate<K.revision_match_rate_threshold){
      await enter('REVISION');
      let it=0;const t0=Date.now();
      while(S.matchRate<K.revision_match_rate_threshold&&it<K.revision_iteration_cap){
        const old=S.cfg.tolerance;S.cfg.tolerance=Math.round(old*1.2*10000)/10000;S.rev.tol=S.cfg.tolerance;
        S.exec=scoreAll(S.tables[S.cfg.left_table],S.tables[S.cfg.right_table],S.cfg);
        computeMetrics();it++;S.rev.it=it;S.cost+=0.004;
        S.revHistory.push({it,tol:S.cfg.tolerance,rate:S.matchRate,note:'safe tweak ±20%'});
        logP2('POLICY_REVISED','iter '+it+' tolerance '+S.cfg.tolerance+' → rate '+S.matchRate.toFixed(3));
        renderAllBodies();await sleep(500);
      }
      if(S.matchRate<K.revision_match_rate_threshold){
        const v=await chatAsk('Stopping revision loop: '+it+' iterations reached without convergence. Best match rate '+
          (S.matchRate*100).toFixed(0)+'%. Resume to continue to QA with the best result.',
          [{label:'Proceed to QA with best result',val:'proceed'},{label:'Abort run',val:'abort'}]);
        decision({state:'REVISION',actor:'user',decision_kind:'revision_halt_ack',proposal:{pair:'halt'},final:{pair:v},confidence:S.matchRate});
        if(v==='abort')requestAbort();
        ck();
      }
    }
    await enter('QA');
    S.queue=[];S.dups=S.exec.dups;
    const rowsL=S.tables[S.cfg.left_table],rowsR=S.tables[S.cfg.right_table];
    S.exec.uctx.forEach(u=>{
      const ctx=buildCtx(u,rowsL,rowsR,S.cfg);
      u.rec.reason=classify(u.rec,ctx);
      const conf=excConfidence(u.ev.length,u.rec.reason);
      const action=decideAction(conf,u.ev.length,u.rec.reason);
      S.queue.push({rec:u.rec,ctx,pieces:u.ev,conf,action,explanation:explain(u.rec,ctx)});
      decision({state:'QA',actor:'system',decision_kind:'exception_classify',proposal:{pair:u.rec.reason},final:{pair:action},confidence:conf});
    });
    renderAllBodies();
    chatPost('QA classified '+S.queue.length+' exceptions by fixed priority order: '+
      S.queue.filter(q=>q.action==='auto_resolve').length+' auto-approve, '+
      S.queue.filter(q=>q.action==='request_confirmation').length+' need review.',[{label:'Exception Queue',panel:'exceptions',primary:true}]);

    await enter('RESOLVING');
    chatPost('Auto-approvals recorded as NO ERROR. Review the rest in the Exception Queue, then finalize.',[{label:'Open Exception Queue',panel:'exceptions',primary:true}]);
    await gate('finalize');ck();
    S.finalized=true;
    decision({state:'RESOLVING',actor:'user',decision_kind:'finalize',proposal:{pair:'keep open'},final:{pair:'finalized'},confidence:1});

    await enter('AGGREGATING');
    buildFinal();renderAllBodies();
    chatPost('Aggregating financial balances and signing the audit chain…');

    await enter('ARCHIVED');
    logP2('SANDBOX_DESTROYED','sbx-demo · mounts removed');
    logP2('RUN_COMPLETE','report + audit retained indefinitely');
    S.state='ARCHIVED';setChips();
    chatPost('Reconciliation complete: '+(S.final.match_rate*100).toFixed(0)+'% matched, '+S.final.honest_exception_count+
      ' exceptions queued, '+S.final.auto_resolved_count+' auto-resolved.',
      [{label:'Results panel',panel:'results',primary:true},{label:'Audit chain',panel:'audit'}]);
    chatPost('Run archived. Ask me about any order (e.g. “ORD_2”), fees, or the exception queue.');
  }catch(e){
    if(e.abort){
      S.state='ABORTED';setChips();
      logP2('ABORT_CONFIRMED','token '+S.abortToken);
      chatPost('Run aborted at '+S.state+'. Session state retained — resume with Start run.');
    } else {S.state='RUN_FAILED';setChips();chatPost('Run failed: '+e.message);logP2('RUN_FAILED',e.message);}
  }
  S.running=false;$('#btnAbort').disabled=true;renderAllBodies();
}
function computeMetrics(){
  const m=S.exec.matched.length,u=S.exec.unmatched.length;
  S.matchRate=m/Math.max(m+u,1);
  const nrows=Object.values(S.tables).reduce((s,t)=>s+t.length,0);
  S.throughput=nrows/Math.max((Date.now()-S.startedAt)/1000,0.001);
  if(S.truth.length){
    const pred=new Set(S.exec.matched.map(x=>x.l_rid+'-'+x.r_rid));
    const tr=new Set(S.truth.map(t=>t[0]+'-'+t[1]));
    let inter=0;pred.forEach(p=>{if(tr.has(p))inter++;});
    S.precision=pred.size?inter/pred.size:null;S.recall=tr.size?inter/tr.size:null;
  }
}
function buildFinal(){
  const rowsL=S.tables[S.cfg.left_table],rowsR=S.tables[S.cfg.right_table];
  const g=rowsL.reduce((s,r)=>s+(+r[S.cfg.left_amount]||0),0);
  const n=rowsR.reduce((s,r)=>s+(+r[S.cfg.right_amount]||0),0);
  const mrids=new Set(S.exec.matched.map(m=>m.l_rid));
  const mv=rowsL.filter(r=>mrids.has(r._rid)).reduce((s,r)=>s+(+r[S.cfg.left_amount]||0),0);
  S.totals={gross:g,net:n,fees:g-n,matched_value:mv,exception_value:g-mv};
  S.final={
    match_rate:S.matchRate,precision_vs_truth:S.precision,recall_vs_truth:S.recall,
    throughput_rows_per_sec:S.throughput,honest_exception_count:S.queue.length,
    auto_resolved_count:S.queue.filter(q=>['auto_resolve','mark_resolved'].includes(q.action)).length,
    escalated_count:S.queue.filter(q=>['request_confirmation','escalate'].includes(q.action)).length,
    unresolved_count:S.queue.filter(q=>q.action==='mark_pending').length,
    total_gross:S.totals.gross,total_net:S.totals.net,total_fees:S.totals.fees,
    matched_value:S.totals.matched_value,exception_value:S.totals.exception_value,
    cost_usd:S.cost,elapsed_seconds:(Date.now()-S.startedAt)/1000,
    llm_user_disagreements:S.audit.filter(r=>r.payload.event==='DECISION'&&r.payload.overridden)
      .map(r=>({seq:r.seq,kind:r.payload.decision_kind,proposal:r.payload.proposal,final:r.payload.final})),
    fallback_events:S.fb.slice(),constants_version:'v0',
    retention_note:'intermediates 90d; final report + audit retained indefinitely',storage_backend:'local_hash_chain'
  };
}

function handleText(t){
  userLine(t);const q=t.toLowerCase();
  if(q==='help')return void chatPost('Commands: samples, start, status, abort, show <panel>. After a run, ask about orders (ORD_2), fees, balances, or exceptions.');
  if(q==='samples')return void loadSamples();
  if(q==='start')return void runPipeline();
  if(q==='abort')return void requestAbort();
  if(q==='status')return void chatPost('State '+S.state+' · breaker '+S.breaker+'/3 · cost $'+S.cost.toFixed(4)+(S.matchRate!=null?' · match rate '+(S.matchRate*100).toFixed(0)+'%':'')+'.');
  if(q.startsWith('show ')||q.startsWith('open ')){
    const name=q.replace(/^(show|open)\s+/,'').trim();
    const id=Object.keys(PANELS).find(p=>p.startsWith(name)||PANELS[p].title.toLowerCase().startsWith(name));
    if(id){focusPanel(id);return void chatPost('Focused the '+PANELS[id].title+' panel.');}
    return void chatPost('Unknown panel: '+name+'. Panels: '+Object.keys(PANELS).join(', ')+'.');
  }
  if(S.final||S.queue.length){
    const up=t.toUpperCase();
    const ref=up.match(/[A-Z]{2,}_?\d+/);
    if(ref){
      const key=ref[0];
      const lrow=Object.values(S.tables)[0].find(r=>String(r[S.cfg?S.cfg.left_key:'order_id'])===key);
      const rrow=Object.values(S.tables)[1].find(r=>String(r[S.cfg?S.cfg.right_key:'utr'])===key);
      const exc=S.queue.find(x=>x.rec.ref===key);
      if(lrow||rrow||exc){
        let msg='Reference '+key+': ';
        if(lrow)msg+='ledger '+money(lrow[S.cfg.left_amount])+' on '+lrow[S.cfg.left_date]+'. ';
        if(rrow)msg+='bank credit '+money(rrow[S.cfg.right_amount])+' on '+rrow[S.cfg.right_date]+'. ';
        if(exc)msg+='Exception: '+exc.rec.reason+' — '+exc.explanation;
        else if(lrow&&rrow)msg+='Matched within tolerance.';
        return void chatPost(msg,[{label:'Exception Queue',panel:'exceptions'}]);
      }
      return void chatPost('Reference '+key+' is not in the active session dataset.');
    }
    if(q.includes('fee'))return void chatPost('Fee model: razorpay_test_mode v1.0 — 2% MDR plus 18% GST. Example: gross ₹2,000.00 → fee ₹47.20 → net settlement ₹1,952.80 (this explains the ORD_2 match).');
    if(q.includes('balance')||q.includes('total')||q.includes('gross'))return void chatPost(S.totals
      ?'Gross ledger '+money(S.totals.gross)+' · net bank inflow '+money(S.totals.net)+' · fees variance '+money(S.totals.fees)+' · matched value '+money(S.totals.matched_value)+' · exception value '+money(S.totals.exception_value)+'.'
      :'Balances are computed at AGGREGATING — run the pipeline first.',[{label:'Results panel',panel:'results'}]);
    if(q.includes('exception'))return void chatPost(S.queue.length
      ?'Queue: '+S.queue.length+' exceptions — '+S.queue.filter(x=>x.action==='auto_resolve').length+' auto-approved (no error), '+
        S.queue.filter(x=>x.action==='request_confirmation').length+' awaiting review, '+S.queue.filter(x=>x.action==='mark_pending').length+' pending.',[{label:'Exception Queue',panel:'exceptions',primary:true}]
      :'No exceptions yet.');
    if(q.includes('audit')||q.includes('chain'))return void chatPost('Audit chain: '+S.audit.length+' records · SHA-style hash chain '+(auditVerify()?'VERIFIED':'FAILED')+'. Append-only; disagreements preserved.',[{label:'Audit panel',panel:'audit'}]);
  }
  if(S.running)chatPost('A run is in state '+S.state+'. Approvals appear here; artifacts render in their panels.');
  else if(!Object.keys(S.tables).length)chatPost('Register data first — say “samples” or upload two CSVs.');
  else chatPost('Ready. Say “start” to run the state machine.');
}
